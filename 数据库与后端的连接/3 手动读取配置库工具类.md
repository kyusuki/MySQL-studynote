如果未采用`Spring Boot`的自动配置，那就可以手动写一个`DBHelper`类配置工具类


基于`HikariCP`数据库连接池的`JDBC`工具类，统一管理数据库连接
1.所需要的依赖包
```
import com.zaxxer.hikari.HikariConfig; 
import com.zaxxer.hikari.HikariDataSource; 

import java.sql.*; 
import java.io.InputStream; 
import java.util.Properties; 

import org.slf4j.Logger; //这个是和日志相关的包
import org.slf4j.LoggerFactory;
```

2.主代码部分
```
public class DBHelper {  
    
    /**类定义与常量**/
    private static final Logger log=LoggerFactory.getLogger(DBHelper.class); 
    //添加连接池做链接优化  
    private static HikariDataSource dataSource;  
    
    /**静态代码块，初始化连接池**/
    static{  
        try{  
            //读取类路径下的db.properties配置文件
            InputStream is=DBHelper.class.getResourceAsStream("/db.properties");  
            Properties properties=new Properties();  
            properties.load(is);  
            
            //创建连接池配置对象
            HikariConfig config=new HikariConfig();  
            
            //从配置文件中读取数据库信息
            config.setJdbcUrl(properties.getProperty("db.url"));  
            config.setUsername(properties.getProperty("db.username"));  
            config.setPassword(properties.getProperty("db.password"));  
            
            //连接池优化核心参数配置
            config.setMaximumPoolSize(10);  
            config.setMinimumIdle(2);  
            config.setConnectionTimeout(30000);  
            
            //创建连接池实例
            dataSource=new HikariDataSource(config);  
        }catch (Exception e){  
            log.error("数据库连接错误",e);  
        }  
    }  
    
    /**获取连接的方法**/
    public static Connection getConnection() throws SQLException{  
        return dataSource.getConnection();  
    }  
    
    /**关闭资源的方法**/
    public static void close(Connection conn, Statement stmt, ResultSet rs) {  
        try { if (rs != null) rs.close(); } catch (Exception e) {}  
        try { if (stmt != null) stmt.close(); } catch (Exception e) {}  
        try { if (conn != null) conn.close(); } catch (Exception e) {}  
    }  
}
```
